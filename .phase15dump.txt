325:     throw new Error(`Verification update failed: ${profileError.message}`);

339: /**
340:  * Generate a secure 1-hour signed URL for viewing/downloading a contractor's resume.
341:  */
342: export async function getSignedResumeUrl(resumePathOrUrl: string): Promise<string> {
343:   if (!resumePathOrUrl) {
344:     throw new Error('No resume file path provided.');
345:   }
346: 
347:   // If it's already a full HTTP URL, check if we need to sign a path
348:   if (resumePathOrUrl.startsWith('http://') || resumePathOrUrl.startsWith('https://')) {
349:     // Extract storage path if it's from contractor-resumes bucket
350:     const match = resumePathOrUrl.match(/\/contractor-resumes\/(.+)$/);
351:     if (match && match[1]) {
352:       const storagePath = decodeURIComponent(match[1]);
353:       const { data, error } = await supabase.storage
354:         .from('contractor-resumes')
355:         .createSignedUrl(storagePath, 3600); // 1 hour validity
356: 
357:       if (!error && data?.signedUrl) {
358:         return data.signedUrl;
359:       }
360:     }
361:     return resumePathOrUrl;
362:   }
363: 
364:   // Generate signed URL from storage path
365:   const { data, error } = await supabase.storage
366:     .from('contractor-resumes')
367:     .createSignedUrl(resumePathOrUrl, 3600);
368: 
369:   if (error || !data?.signedUrl) {
370:     console.warn('[contractorService] getSignedResumeUrl failed, falling back to public URL:', error?.message);
371:     const { data: publicData } = supabase.storage
372:       .from('contractor-resumes')
373:       .getPublicUrl(resumePathOrUrl);
374:     return publicData.publicUrl;
375:   }
376: 
377:   return data.signedUrl;
378: }

582:                     disabled={actionBusy}
583:                     onClick={() => void handleApprove(selected)}
584:                     className="px-4 py-2 rounded-xl bg-emerald-600 text-white text-xs font-bold hover:bg-emerald-700 transition-all inline-flex items-center gap-1.5 disabled:opacity-50"
585:                   >
586:                     <Check className="w-3.5 h-3.5" /> Approve Contractor
587:                   </button>
588:                 )}
589:                 <button
590:                   type="button"
591:                   disabled={actionBusy}
592:                   onClick={() => {
593:                     setNextStatus('REJECTED');
594:                     setChangeError(null);
595:                     setChangeOpen(true);
596:                   }}
597:                   className="px-4 py-2 rounded-xl bg-red-600 text-white text-xs font-bold hover:bg-red-700 transition-all inline-flex items-center gap-1.5 disabled:opacity-50"

599:                   <X className="w-3.5 h-3.5" /> Reject
600:                 </button>
601:                 <button
602:                   type="button"
603:                   disabled={actionBusy}
604:                   onClick={() => {
605:                     setNextStatus(isVerificationVerified(selected.verification_status) ? 'PENDING' : 'VERIFIED');
606:                     setChangeError(null);
607:                     setChangeOpen(!changeOpen);
608:                   }}
609:                   className="px-4 py-2 rounded-xl border border-neutral-200 text-neutral-700 text-xs font-bold hover:bg-neutral-50 transition-all inline-flex items-center gap-1.5 disabled:opacity-50"

611:                   <ArrowRight className="w-3.5 h-3.5" /> Change Status
612:                 </button>
613:               </div>
614:               {/* Change status form */}
615:               {changeOpen && (
616:                 <div className="p-4 rounded-2xl bg-neutral-50 border border-neutral-200 space-y-3">
617:                   <label className="text-[10px] font-bold text-neutral-400 uppercase block">Change verification status</label>
618:                   <select
619:                     value={nextStatus}
620:                     onChange={e => setNextStatus(e.target.value as 'VERIFIED' | 'REJECTED' | 'PENDING')}
621:                     className="w-full px-3 py-2 rounded-xl border border-neutral-200 text-xs font-semibold text-neutral-800 bg-white cursor-pointer"
622:                   >
623:                     <option value="VERIFIED">Verified (recommended to customers)</option>
624:                     <option value="REJECTED">Rejected</option>
625:                     <option value="PENDING">Pending</option>
626:                   </select>
627: 
628:                   {nextStatus === 'REJECTED' && (
629:                     <textarea
630:                       value={rejectReason}
631:                       onChange={e => setRejectReason(e.target.value)}

635:                     />
636:                   )}
637: 
638:                   {changeError && (

640:                   )}
641: 
642:                   <button
643:                     type="button"
644:                     disabled={actionBusy}
645:                     onClick={() => void handleChangeStatus(selected)}
646:                     className="w-full py-2.5 rounded-xl bg-neutral-900 text-white text-xs font-bold hover:bg-neutral-800 transition-all inline-flex items-center justify-center gap-1.5 disabled:opacity-50"

664:                 Close
665:               </button>
666:             </div>

670:     </div>
